\section{Motivation}

\begin{frame}{Problem Setting}
    \small
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{block}{Why MicroRTS?}
                \begin{itemize}
                    \item simplified RTS environment for RL research
                    \item still contains resource management, unit production and combat
                    \item supports controlled and repeatable experiments
                \end{itemize}
            \end{block}
            \begin{block}{Main challenge}
                \begin{itemize}
                    \item large combinatorial action space
                    \item simultaneous decisions
                    \item delayed and sparse rewards
                    \item robustness against diverse opponents
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.45\textwidth}
            \includegraphics[width=0.94\linewidth]{initial_pos_basesWorkers16x16A.png}
            \vspace{0.2em}

            \scriptsize Initial state of the \texttt{basesWorkers16x16A} map.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Motivation and Research Question}
    \begin{block}{Starting point}
        The existing Base Agent trained with Behavioral Cloning (BC) and PPO is already strong, but it still has weaknesses against several opponents and a limited strategic repertoire.
    \end{block}
    \begin{block}{Goal}
        Improve robustness and generalization beyond a fixed set of training bots by adding Self-Play and League Training (LT).
    \end{block}
    \begin{alertblock}{Research question}
        How effective is League Training with PPO in improving the robustness and generalization of a BC/PPO-pretrained MicroRTS agent, and how strongly does this depend on the diversity of the initial agents and BC experts?
    \end{alertblock}
\end{frame}

\section{Approach}

\begin{frame}{Training Pipeline}
    % TODO (sagen): The first three phases already existed in the previous work; LT is the main extension of this thesis.
    \begin{center}
        \Large
        BC $\rightarrow$ BC-FT $\rightarrow$ PPO $\rightarrow$ LT $\rightarrow$ PPO fine-tuning
    \end{center}
    \vspace{1em}
    \begin{itemize}
        \item \textbf{BC / BC-FT:} imitate strong scripted opponents to obtain a competent initial policy
        \item \textbf{PPO:} refine the policy through exploration
        \item \textbf{LT:} train against a population of evolving opponents instead of only bots or pure self-play
        \item \textbf{Fine-tuning:} fix remaining weaknesses after LT, especially against WorkerRushAI
    \end{itemize}
\end{frame}

\begin{frame}{From Self-Play to League Training}
    % TODO (sagen): The methods build on each other; each step addresses a concrete limitation of the previous one.
    \small
    \renewcommand{\arraystretch}{1.15}
    \begin{tabularx}{\linewidth}{lX}
        \textbf{SP} & train against the current version of the agent; simple, but prone to overfitting or cyclical strategies \\
        \textbf{FSP} & train against a set of past policies; reduces forgetting, but opponent choice is still crude \\
        \textbf{PFSP} & sample opponents based on current win rates; focuses on informative and non-trivial matchups \\
        \textbf{LT} & add role-based agents: Main Agent, Historical Agents, Main Exploiters and League Exploiters \\
    \end{tabularx}
    \vspace{0.8em}
    \begin{block}{Why LT?}
        LT does not only preserve diversity through historical snapshots, it also creates diversity through exploiters that actively search for weaknesses of the current policy.
    \end{block}
\end{frame}

\begin{frame}{League Training Design}
    % TODO (sagen): The different roles mainly differ in who they train against and when they are checkpointed or reset.
    \begin{itemize}
        \item \textbf{Main Agent:} primary policy
        \item \textbf{Historical Agents:} frozen snapshots for stability
        \item \textbf{Main Exploiters:} exploit weaknesses of historical Main Agents
        \item \textbf{League Exploiters:} search for broader league weaknesses
        \item \textbf{PFSP:} opponent sampling based on current win rates
    \end{itemize}
    \vspace{0.6em}
    \begin{block}{Key idea}
        LT turns robustness against diverse opponents into part of the optimization target.
    \end{block}
\end{frame}

\begin{frame}{Important Implementation Changes}
    % TODO (sagen): These were the main engineering changes needed to make LT work reliably in this codebase.
    \begin{itemize}
        \item dynamic bot environments for CoacAI and Mayari to stabilize the early LT phase
        \item observation adjustment for player 2 in self-play by flipping GridNet observations
        \item delta score reward to provide denser learning signals in MicroRTS
        \item annealed entropy coefficient to increase early exploration, especially for exploiters
        \item new initial exploiter agents with different unit-type focuses to improve diversity
        \item corrected exploiter reset and checkpoint logic compared to the AlphaStar pseudocode
    \end{itemize}
\end{frame}

\begin{frame}{Why New Initial Agents Matter}
    % TODO (sagen): The diversity of the initial agents matters more than just their raw strength.
    % TODO (sagen): New initial agents let exploiters search from different strategic starting points instead of always from the same heavy-rush bias.
    \begin{itemize}
        \item exploiters initialized from similar agents tend to find similar weaknesses
        \item new initial agents focused on different unit types broaden the exploiter set
        \item this improves generalization against a wider range of opponents
        \item LT therefore depends strongly on the diversity already present at initialization
    \end{itemize}
    % TODO: add the exploiter diversity schematic from BA?
\end{frame}

\begin{frame}{Scope and Remaining Limitations}
    % TODO (sagen): Claims about robustness and generalization should be interpreted within this concrete scenario.
    % TODO (sagen): In the previous work, pure PPO used only 24 parallel environments; LT is much more expensive.
    \begin{itemize}
        \item evaluation is limited to one map: \texttt{basesWorkers16x16A}
        \item the setting uses full observability and no fog of war
        \item LT requires substantially more computation than pure PPO
        \item exploiter diversity is still the main bottleneck
    \end{itemize}
    \begin{block}{Takeaway}
        The method works, but the strength of the final result depends strongly on the diversity and cost of the training setup.
    \end{block}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
    % TODO (sagen): A narrow BC expert set may be a suboptimal starting point for LT because it narrows the strategic repertoire too early.
    \begin{block}{Conclusion}
        League Training with PPO improves the robustness and generalization of the MicroRTS Base Agent. Additional PPO fine-tuning improves the final agent even further.
    \end{block}
    \begin{block}{Key takeaway}
        The effectiveness of LT depends strongly on exploiter diversity,5 which in turn depends on the diversity of the initial agents and BC experts.
    \end{block}
    \begin{itemize}
        \item LT improves most matchups and the overall performance
        \item fine-tuning further improves the final agent
        \item diversity of initial agents is the critical practical factor
    \end{itemize}
\end{frame}

\begin{frame}{Future Work}
    % TODO (sagen): The most promising directions are map diversity and stronger diverse initial agents.
    % TODO (sagen): Another practical direction is a faster environment implementation, ideally on the GPU.
    \begin{itemize}
        \item train on multiple maps and with fog of war
        \item improve the diversity and strength of initial agents
        \item reduce computational cost of MicroRTS training
        \item investigate stronger diversity mechanisms for exploiters
    \end{itemize}
\end{frame}
