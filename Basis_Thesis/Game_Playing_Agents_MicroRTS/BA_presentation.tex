\section{Motivation}



\begin{frame}{Goals and Research Question}
    \small
    \begin{block}{Goals}
        \begin{itemize}
            \item improve the Base Agent with League Training (LT)
            \item training against a more diverse set of opponents
            \item focus on robustness and generalization
        \end{itemize}
    \end{block}
    \vspace{0.4em}
    \begin{minipage}{0.94\linewidth}
        \footnotesize
        \textbf{Context: Base Agent}
        \begin{itemize}
            \item combination of Behavioral Cloning (BC) and Proximal Policy Optimization (PPO)
            \item strong baseline, but weaknesses against several opponents
        \end{itemize}
    \end{minipage}
    \vspace{0.2em}
    \begin{alertblock}{Research Questions}
        \begin{itemize}
            \item How much does LT with PPO improve robustness and generalization of the BC/PPO-pretrained agent?
            \item How strongly does this depend on the diversity of initial agents and BC experts?
        \end{itemize}
    \end{alertblock}
\end{frame}


\begin{frame}{Setting}
    \small
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{block}{MicroRTS}
                \begin{itemize}
                    \item simplified RTS environment for RL research
                    \item still contains resource management, unit production and combat
                    \item supports controlled and repeatable experiments
                \end{itemize}
            \end{block}
            \begin{block}{Main RL challenges}
                \begin{itemize}
                    \item large combinatorial action space
                    \item simultaneous decisions
                    \item delayed and sparse rewards
                    \item robustness against diverse opponents
                \end{itemize}
            \end{block}
            \vspace{0.4em}
            \begin{center}
                \href{run:images/MicroRTS_example.mp4}{\beamergotobutton{Open MicroRTS Example Video}}
            \end{center}
        \end{column}
        \begin{column}{0.45\textwidth}
            \includegraphics[width=0.94\linewidth]{initial_pos_basesWorkers16x16A.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Evidence}
    \small
    \begin{block}{Is the improvement really caused by LT, or just by more training?}
        \begin{itemize}
            \item LT alone improves the overall win rate from \textbf{76.5\%} to \textbf{79.2\%}.
            \item LT + FT reaches \textbf{85.7\%}.
            \item As a PPO-only reference, the prior BC/PPO thesis reached about \textbf{79\%} and reported no improvement from further PPO training.
            \item The prior work also attempted a similar setup as PPO-fine-tuning, but couldn't achieve improvements to the Base Agent.
        \end{itemize}
    \end{block}
    \begin{block}{What is the strongest evidence that LT changes the policy?}
        \begin{itemize}
            \item MA after LT beats the Base Agent: \textbf{89\%} win, \textbf{9\%} draw, \textbf{2\%} loss.
            \item Large gains against Mayari, RandomBiasedAI, Rojo and GuidedRojoA3N.
            \item In the render of the evaluations the MA has at least 1 heavy unit actively blocking the opponent's Units in front of the Base.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{WorkerRushAI}
    \small
    \begin{block}{Why does LT perform worse against WorkerRushAI?}
        \begin{itemize}
            \item WorkerRushAI exposes a weakness not sufficiently found by the exploiters.
            \item The Base Agent is biased toward a heavy-rush strategy.
            \item The worker-focused initial agent for the exploiters wasn't strong enough to find it. Both of the HA created got stuch in a local optimum of drawing the Game instead.
        \end{itemize}
    \end{block}
    \begin{block}{Why does PPO fine-tuning repair it?}
        \begin{itemize}
            \item WorkerRushAI is added as a direct training opponent.
            \item The win rate recovers from \textbf{35.0\%} after LT to \textbf{99.5\%}.
            \item Other results mostly remain stable or improve, so no strong overfitting is visible in the evaluated set.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Generalization, Reliability}
    \small
    \begin{block}{Can we call this generalization with only one map?}
        \begin{itemize}
            \item Only within the evaluated setting: one map, full observability and fixed rules.
            \item It is generalization across opponent types, not across maps.
            \item Several strong opponents were not used in training, (e. g. all MCTS-based bots).
        \end{itemize}
    \end{block}
    \begin{block}{What would be needed for a stronger claim?}
        \begin{itemize}
            \item evaluation on multiple maps
            \item fog of war
            \item multiple training seeds
            \item same-compute PPO and ablation baselines
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{PPO and AlphaStar}
    \small
    \begin{block}{Why use PPO instead of AlphaStar's V-trace setup?}
        \begin{itemize}
            \item availability
            \item scope of this thesis
            \item V-trace would likely scale better in a large distributed setup, but is more complex.
        \end{itemize}
    \end{block}
    \begin{block}{What is different from AlphaStar?}
        \begin{itemize}
            \item compute budget and scale
            \item PPO instead of off-policy actor-critic with V-trace
            \item initial policy: BC/PPO bots, not from broad human replay data
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Diversity}
    \small
    \begin{block}{Why is exploiter diversity the main limitation?}
        \begin{itemize}
            \item LT can only improve weaknesses that exploiters actually find.
            \item Most league agents start with the Base Agent policy.
            \item The Base Agent is strong, but biased toward a heavy-rush strategy.
            \item Very different exploits require first unlearning this default behavior to be explored.
            \item During this transition the exploiter performs poorly, so PPO gets weak learning signals.
            \item Therefore exploiters tend to find similar weaknesses instead of covering all relevant strategies.
            \item WorkerRushAI shows that some important weaknesses were not found by the exploiters.
        \end{itemize}
    \end{block}
    \begin{block}{Why add new initial agents?}
        \begin{itemize}
            \item They give exploiters different strategic starting points.
            \item This improves the exploiters ability, to reach the MAs weaknesses that are only exploitable with very different strategies as the Base Agents.
            \item From prior runs we saw, that reaching strategies of different unit types has seemingly the biggest difficulties.
            \item It supports the hypothesis that BC expert diversity strongly affects LT performance.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Bot Environments}
    \small
    \begin{block}{Why keep CoacAI and Mayari during LT?}
        \begin{itemize}
            \item They stabilize the early LT phase before exploiters find useful weaknesses.
            \item They prevent the Main Agent from drifting away from already strong behavior.
            \item The number of bot environments is dynamic, so they become less dominant later.
        \end{itemize}
    \end{block}
    \begin{block}{Does this make the evaluation unfair?}
        \begin{itemize}
            \item CoacAI and Mayari are training opponents, so they are not the main generalization evidence.
            \item stronger evidence comes from opponents not used in training.
            \item The prior work showed that improving against CoacAI and Mayari with pure PPO is difficult.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Player Side}
    \small
    \begin{block}{Why adjust observations for Player 2?}
        \begin{itemize}
            \item The Base Agent was trained only as Player 1.
            \item Gym-MicroRTS: Toward affordable full game real-time strategy games research with deep reinforcement learning implemented SP and concluded that a likely reason for worse performance of SP was that SP also has to train, to start as Player 2, while pure PPO only trained as Player 1.
            \item difficulties retrainig the Base Agent.
            \item Player-2 observations, action masks and action directions are flipped.
            \item This lets the same policy play both sides without relearning the map orientation from scratch.
        \end{itemize}
    \end{block}
    \begin{block}{Does Player 1 priority bias the results?}
        \begin{itemize}
            \item It is a potential bias in conflict resolution.
            \item Evaluations of the Base Agent had the same bias making the comparison fair.
            \item P1: LT-FT against P2: Base Agent over 100 games: win rate 85\%, draw rate 15\%, loss rate 0\%
            \item P2: LT-FT against P1: Base Agent over 100 games: win rate 89\%, draw rate 11\%, loss rate 0\%
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Draws}
    \small
    \begin{block}{Why use loss rate in addition to win rate?}
        \begin{itemize}
            \item Draw-heavy opponents can hide defensive improvements in the win rate.
            \item The final loss rate drops from \textbf{7.7\%} to \textbf{2.2\%}.
            \item This suggests more robust defense, not only stronger attacks.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Statistical Reliability}
    \small
    \begin{block}{Are the improvements statistically reliable?}
        \begin{itemize}
            \item Large gains are unlikely to be explained only by evaluation noise.
            \item Small differences, e. g. overall LT vs Base Agent, should be interpreted cautiously.
            \item most of the multiple training runs with slightly different hyperparameters showed similar improvements from LT, but the exact values varied.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Why Not Fewer Environments?}
    \begin{itemize}
        \item The number of environments affects rollout size.
        \item Smaller rollouts make PPO updates more unstable because they provide less information about the update direction.
        \item More steps per rollout create similar tradeoffs to adding more environments.
    \end{itemize}
